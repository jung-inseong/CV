%% 자동 번호 항목.
%% 클래스가 제공하는 \NumberedItem{라벨} 은 라벨을 직접 넘겨야 해서
%% 항목을 넣고 뺄 때마다 번호를 손으로 고쳐야 한다. 카운터를 물려
%% \NumItem 하나로 자동으로 매겨지게 감쌌다.
%%
%% \MaxNumberedItem 은 번호 자리의 너비 기준이다. 두 자리(88.)로
%% 잡아두면 항목이 10개를 넘어도 들여쓰기가 흔들리지 않는다.
\renewcommand{\MaxNumberedItem}{88.}
\newcounter{PresNo}
\newcommand{\ResetNumbering}{\setcounter{PresNo}{0}}
\newcommand{\NumItem}{\refstepcounter{PresNo}\NumberedItem{\thePresNo.}}

%% 항목 들여쓰기 통일.
%% 번호 항목은 두 자리 숫자 자리를 잡느라 불릿 항목보다 본문이 더
%% 오른쪽에서 시작한다. 그대로 두면 섹션마다 본문 시작 위치가 어긋나므로,
%% 번호 항목의 열 너비(\CVItemCol)를 기준으로 나머지를 모두 맞춘다.
%% 폰트가 정해진 뒤에 재야 하므로 \AtBeginDocument 에서 측정한다.
\newlength{\CVItemCol}
\AtBeginDocument{%
\settowidth{\CVItemCol}{\hspace{1.5mm}\MaxNumberedItem\hspace{2.5mm}}}

%% 불릿 항목: 기호를 번호와 같은 열에 우측 정렬한다.
\renewcommand{\BulletItem}
{\par\vspace{0\baselineskip}%
\savebox{\BulletItemIndentBox}{\makebox[\CVItemCol][r]{\BulletSymbol\hspace{2.5mm}}}%
\settowidth{\BulletItemIndentWidth}{\usebox{\BulletItemIndentBox}}%
\parshape 2%
\labelwidth \linewidth%
\dimexpr\labelwidth+\BulletItemIndentWidth\relax \dimexpr\linewidth-\BulletItemIndentWidth\relax%
\usebox{\BulletItemIndentBox}%
\ignorespaces%
}

%% 기호 없이 같은 열에서 시작하는 항목 (Research Interests 등).
\renewcommand{\Item}
{\par\vspace{0\baselineskip}%
\savebox{\BulletItemIndentBox}{\hspace{\CVItemCol}}%
\settowidth{\BulletItemIndentWidth}{\usebox{\BulletItemIndentBox}}%
\parshape 2%
\labelwidth \linewidth%
\dimexpr\labelwidth+\BulletItemIndentWidth\relax \dimexpr\linewidth-\BulletItemIndentWidth\relax%
\usebox{\BulletItemIndentBox}%
\ignorespaces%
}

%% 하위 항목: 위 열에서 한 단계 더 들어간다.
\renewcommand{\SubBulletItem}
{\par\vspace{0\baselineskip}%
\savebox{\SubBulletItemIndentBox}{\hspace{\dimexpr\CVItemCol+1.5mm\relax}%
\SubBulletSymbol\hspace{1.5mm}}%
\settowidth{\SubBulletItemIndentWidth}{\usebox{\SubBulletItemIndentBox}}%
\parshape 2%
\labelwidth \linewidth%
\dimexpr\labelwidth+\SubBulletItemIndentWidth\relax \dimexpr\linewidth-\SubBulletItemIndentWidth\relax%
\usebox{\SubBulletItemIndentBox}%
\ignorespaces%
}
